\section{Principios}
\begin{enumerate}[leftmargin=*]
\item \textbf{Ningún cargo para los autores.} Desde Cuba no es posible pagar cargos por procesamiento
  de artículos (APC). Se eligen revistas \emph{híbridas}, en las que la ruta de suscripción no tiene
  coste para los autores y el acceso abierto es opcional, o revistas de acceso abierto
  \emph{diamante}, sin cargos para autores ni lectores.
\item \textbf{Visibilidad sin pagar.} La ruta de suscripción se complementa con un preprint (por
  ejemplo, en medRxiv) y con el depósito del manuscrito aceptado según la política de cada editorial,
  además del repositorio público del código y del release con control de divulgación.
\item \textbf{Ajuste revista--artículo.} El artículo~1 es un análisis de desproporcionalidad que
  sigue READUS-PV, guía publicada en \emph{Drug Safety}; el artículo~2 vincula calidad de lotes y
  reactogenicidad, tema de \emph{Vaccine} y \emph{Biologicals}.
\item \textbf{Un artículo por revista y envío secuencial.} No se envía el mismo trabajo a dos revistas
  a la vez. El artículo~2 cita el~1 como análisis complementario (``enviado'' o ``en prensa'').
\end{enumerate}

\paragraph{Revistas excluidas.} Se descartan las revistas exclusivamente de acceso abierto con APC
(por ejemplo, \emph{Expert Review of Vaccines}, de acceso abierto completo desde el volumen 22,
2023) salvo que se obtenga una exención total por escrito antes del envío. Cuba figura en el
Grupo~B de la lista de elegibilidad de Research4Life de 2025 (acceso de bajo costo; verificar en el momento del envío), lo que no garantiza exenciones totales de APC.

\section{Secuencia y calendario}
Los plazos son estimaciones para planificar; los tiempos editoriales reales varían.
\begin{enumerate}[leftmargin=*]
\item \textbf{Fase 1 (local).} Ejecución del flujo con los datos reales en la estación controlada
  (\texttt{vamengoc run}), revisión del informe de calidad de datos y del registro de auditoría.
\item \textbf{Fase 2 (publicación).} Verificación del release con control de divulgación
  (\texttt{vamengoc release verify public}), revisión por el responsable de datos y \emph{commit}
  de \nolinkurl{release/public}. Todos los documentos se recompilan automáticamente con los datos reales.
\item \textbf{Auditoría final.} Repetir la auditoría de los manuscritos sobre los números reales,
  completar los campos marcados ``a completar por los autores'' y obtener la aprobación ética.
\item \textbf{Envío del artículo~1} a \emph{Drug Safety} y depósito simultáneo del preprint.
\item \textbf{Envío del artículo~2} a \emph{Vaccine} cuando el~1 esté en revisión.
\item \textbf{Revisión por pares.} Responder punto por punto; cualquier análisis nuevo se ejecuta
  en el flujo para que las cifras sigan generándose automáticamente.
\item \textbf{Tras la aceptación.} Elegir la ruta de suscripción (sin cargos) y depositar el
  manuscrito aceptado según la política de la editorial.
\end{enumerate}

\section{Paquete de envío}
\begin{itemize}[leftmargin=*]
\item Manuscrito con página de título, resumen, palabras clave, declaraciones y referencias
  (\nolinkurl{papers/*/en/manuscript.pdf}).
\item Material suplementario (\nolinkurl{supplement.pdf}) y lista de verificación (READUS-PV para el
  artículo~1; RECORD-PE para el~2).
\item Carta de presentación (\nolinkurl{submission/*/cover_letter.pdf}).
\item \emph{Highlights} para \emph{Vaccine} (\nolinkurl{submission/p2-lot-quality/highlights.pdf});
  \emph{Key Points} para \emph{Drug Safety} (incluidos en el manuscrito).
\item Figuras en PDF vectorial (\nolinkurl{release/public/figures/en}).
\end{itemize}

\section{Pendiente de los autores}
\begin{itemize}[leftmargin=*]
\item Aprobación del comité de ética (nombre, número y fecha) y del responsable de datos.
\item Fuente de financiación o declaración de ausencia de financiación específica.
\item Funciones CRediT de cada autor y correo del autor para la correspondencia.
\item Fecha de extracción de los datos y custodio de la base (en \nolinkurl{configs/pipeline.yaml}).
\item Confirmación del acceso a las plataformas de envío (Editorial Manager, ScholarOne) desde la red
  institucional.
\end{itemize}
